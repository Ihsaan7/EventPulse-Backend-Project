import { ApiError } from "../utils/ApiError.js";

/**
 * @description Middleware to verify if the user has the required role(s)
 * @param {Array<string>} roles - Array of allowed roles (e.g., ['ADMIN', 'ORGANIZER'])
 */
export const verifyRole = (...roles) => {
    return (req, res, next) => {
        // 1. Check if user exists on the request (attached by verifyJWT)
        if (!req.user) {
            throw new ApiError(401, "Unauthorized: User not found");
        }

        // 2. Check if the user's role is included in the allowed roles
        if (!roles.includes(req.user.role)) {
            throw new ApiError(
                403, 
                "Forbidden: You do not have permission to access this resource"
            );
        }

        // 3. If role matches, proceed to the controller
        next();
    };
};
