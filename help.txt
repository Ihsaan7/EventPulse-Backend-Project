import { asyncHandler } from "../utils/asyncHandler.js";
import { ApiError } from "../utils/ApiError.js";
import jwt from "jsonwebtoken";
import { getDB } from "../db/index.js";
import { findUserById } from "../models/user.model.js";

/**
 * @description Middleware to verify JWT and attach user to request
 */
export const verifyJWT = asyncHandler(async (req, res, next) => {
    // 1. Extract token from cookies or Authorization header (Bearer token)
    const token = 
        req.cookies?.accessToken || 
        req.header("Authorization")?.replace("Bearer ", "");

    if (!token) {
        throw new ApiError(401, "Unauthorized request");
    }

    // 2. Verify the token
    let decodedToken;
    try {
        decodedToken = jwt.verify(token, process.env.ACCESS_TOKEN_SECRET);
    } catch (error) {
        throw new ApiError(401, "Invalid or expired Access Token");
    }

    // 3. Find the user in the database
    const db = getDB();
    const user = await findUserById(db, decodedToken._id);

    if (!user) {
        throw new ApiError(401, "Invalid Access Token: User not found");
    }

    // 4. Attach user to request and move to next middleware/controller
    req.user = user;
    next();
});
